-- ============================================================================
-- TEST 1 — IDEMPOTENCY: process_order_rewards gọi 3 lần → tích lũy CHỈ tăng 1 lần
-- ============================================================================
begin;
create temp table _r(test text, ket_qua text, chi_tiet text) on commit drop;
insert into public.orders(id, user_id, status, payment_method, total_amount, note, rewards_processed_at)
values ('a1111111-1111-1111-1111-111111111111',
        '1b660080-2093-4260-9c21-2dadd3aacab6', 'done', 'cod', 300000, '[B11-TEST1]', null);
insert into public.order_items(order_id, product_id, quantity, price_at_time)
values ('a1111111-1111-1111-1111-111111111111',
        '0e6a4ff2-0967-4ae8-9705-855f63e82c59', 1, 250000);
do $$
declare
  v_order uuid := 'a1111111-1111-1111-1111-111111111111';
  v_user uuid; v_before bigint; v_after bigint; v_amount bigint;
begin
  select user_id, total_amount into v_user, v_amount from public.orders where id = v_order;
  select lifetime_spend into v_before from public.profiles where id = v_user;
  perform public.process_order_rewards(v_order);
  perform public.process_order_rewards(v_order);   -- gọi lại
  perform public.process_order_rewards(v_order);   -- gọi lại lần 3
  select lifetime_spend into v_after from public.profiles where id = v_user;
  insert into _r values ('TEST1 idempotency',
    case when v_after = v_before + v_amount then '✅ PASS' else '❌ FAIL' end,
    format('trước=%s, sau=%s, +đơn=%s (kỳ vọng sau=%s)', v_before, v_after, v_amount, v_before+v_amount));
end $$;
select * from _r;     -- 👈 KẾT QUẢ Ở ĐÂY
rollback;             -- xoá đơn TEST1, trả DB về như cũ


-- ============================================================================
-- TEST 2 — TRIGGER chỉ bắt CHUYỂN-CẠNH: set done→done lần nữa KHÔNG phát lại
-- ============================================================================
begin;
create temp table _r(test text, ket_qua text, chi_tiet text) on commit drop;
insert into public.orders(id, user_id, status, payment_method, total_amount, note, rewards_processed_at)
values ('a2222222-2222-2222-2222-222222222222',
        '1b660080-2093-4260-9c21-2dadd3aacab6', 'done', 'cod', 150000, '[B11-TEST2]', now());
do $$
declare
  v_order uuid := 'a2222222-2222-2222-2222-222222222222';
  v_user uuid; v_before bigint; v_after bigint;
begin
  select user_id into v_user from public.orders where id = v_order;
  select lifetime_spend into v_before from public.profiles where id = v_user;
  update public.orders set status = 'done' where id = v_order;  -- done -> done (không phải cạnh)
  select lifetime_spend into v_after from public.profiles where id = v_user;
  insert into _r values ('TEST2 done->done không phát lại',
    case when v_after = v_before then '✅ PASS' else '❌ FAIL' end,
    format('trước=%s, sau=%s (kỳ vọng giữ nguyên)', v_before, v_after));
end $$;
select * from _r;
rollback;


-- ============================================================================
-- TEST 3 — DOUBLE-SPEND voucher qua place_order (đặt 2 lần cùng 1 voucher)
-- ============================================================================
begin;
create temp table _r(test text, ket_qua text, chi_tiet text) on commit drop;
grant insert, select on _r to authenticated;
insert into public.voucher_definitions(id, code, name, type, value, min_order, is_active)
values ('b3333333-3333-3333-3333-333333333333', 'B11T3', 'TEST3 fixed 100k', 'fixed', 100000, 200000, true);
insert into public.customer_vouchers(id, user_id, voucher_def_id, source, status, issued_at, expires_at)
values ('c3333333-3333-3333-3333-333333333333',
        '1b660080-2093-4260-9c21-2dadd3aacab6',
        'b3333333-3333-3333-3333-333333333333', 'manual', 'active', now(), now() + interval '30 days');
set local role authenticated;
select set_config('request.jwt.claims',
       json_build_object('sub','1b660080-2093-4260-9c21-2dadd3aacab6',
                         'email','minhquyet08122003@gmail.com','role','authenticated')::text, true);
do $$
declare
  v_voucher uuid := 'c3333333-3333-3333-3333-333333333333';
  v_items jsonb := jsonb_build_array(
            jsonb_build_object('product_id','0e6a4ff2-0967-4ae8-9705-855f63e82c59','quantity', 3)); -- 750k ≥ min 200k
  v_o1 uuid; v_st text; v_blocked boolean := false;
begin
  v_o1 := public.place_order(v_items, 'cod', 'TEST3 lần 1', v_voucher);   -- lần 1: phải OK
  select status into v_st from public.customer_vouchers where id = v_voucher;
  insert into _r values ('TEST3a đặt đơn + voucher->used',
    case when v_o1 is not null and v_st = 'used' then '✅ PASS' else '❌ FAIL' end,
    format('order=%s, voucher=%s', v_o1, v_st));
  begin
    perform public.place_order(v_items, 'cod', 'TEST3 lần 2', v_voucher);  -- lần 2: phải bị chặn
  exception when others then v_blocked := true;
  end;
  insert into _r values ('TEST3b chặn double-spend',
    case when v_blocked then '✅ PASS' else '❌ FAIL' end,
    case when v_blocked then 'dùng lại bị từ chối (đúng)' else 'LẼ RA phải bị chặn!' end);
end $$;
reset role;
select * from _r;
rollback;


-- ============================================================================
-- TEST 4 — BẢO MẬT RLS: khách thường KHÔNG ghi được bảng cấu hình / ví
--   Dùng KHÁCH THƯỜNG quyetle08122003@gmail.com (KHÔNG phải admin).
-- ============================================================================
begin;
create temp table _r(test text, ket_qua text, chi_tiet text) on commit drop;
grant insert, select on _r to authenticated;
insert into public.voucher_definitions(id, code, name, type, value, min_order, is_active)
values ('b4444444-4444-4444-4444-444444444444', 'B11T4', 'TEST4 def', 'fixed', 50000, 0, true);
set local role authenticated;
select set_config('request.jwt.claims',
       json_build_object('sub','72ee7892-aedc-4d0b-b386-a811571706ac',
                         'email','quyetle08122003@gmail.com','role','authenticated')::text, true);
do $$
declare v_blocked_a boolean := false; v_blocked_b boolean := false; v_rows int := 0;
begin
  -- 4a) Tự phát voucher cho mình → phải bị RLS chặn
  begin
    insert into public.customer_vouchers(user_id, voucher_def_id, source, status)
    values ('72ee7892-aedc-4d0b-b386-a811571706ac',
            'b4444444-4444-4444-4444-444444444444', 'manual', 'active');
  exception when others then v_blocked_a := true;
  end;
  insert into _r values ('TEST4a chặn khách tự phát voucher',
    case when v_blocked_a then '✅ PASS' else '❌ FAIL' end,
    case when v_blocked_a then 'insert bị RLS chặn (đúng)' else 'RLS HỞ — khách ghi được!' end);

  -- 4b) Tự sửa cấu hình hạng → phải bị chặn (raise) hoặc 0 hàng bị tác động
  begin
    update public.membership_tiers set discount_percent = 99 where code = 'member';
    get diagnostics v_rows = row_count;
  exception when others then v_blocked_b := true;
  end;
  insert into _r values ('TEST4b chặn khách sửa membership_tiers',
    case when v_blocked_b or v_rows = 0 then '✅ PASS' else '❌ FAIL' end,
    case when v_blocked_b then 'bị RLS chặn (đúng)'
         when v_rows = 0 then 'update 0 hàng (bị RLS lọc, đúng)'
         else format('RLS HỞ — sửa được %s hàng!', v_rows) end);
end $$;
reset role;
select * from _r;
rollback;

       json_build_object('sub','1b660080-2093-4260-9c21-2dadd3aacab6',

values ('c8888888-8888-8888-8888-888888888888',
        '1b660080-2093-4260-9c21-2dadd3aacab6',
        'b8888888-8888-8888-8888-888888888888', 'manual', 'active', now(), now() + interval '30 days');
